## 2. Auth Store

```typescript
// src/features/auth/store/auth.store.ts
import { create } from 'zustand';
import { Session } from '@supabase/supabase-js';

interface AuthState {
  session: Session | null;
  isLoading: boolean;
  setSession: (session: Session | null) => void;
  setLoading: (isLoading: boolean) => void;
}

export const useAuthStore = create<AuthState>((set) => ({
  session: null,
  isLoading: true, // app শুরুতে true — session check শেষ না হওয়া পর্যন্ত
  setSession: (session) => set({ session }),
  setLoading: (isLoading) => set({ isLoading }),
}));
```

`isLoading` কেন দরকার: app চালু হওয়ার সাথে সাথে আমরা জানি না user logged in কিনা — SecureStore থেকে session read করতে সময় লাগে। এই state না থাকলে splash-এর পরপরই ভুলভাবে login screen flash করে ফেলবে, পরে session পেলে home-এ redirect করবে — খারাপ UX।

## 3. Selector hooks (component-এ ব্যবহার সহজ করার জন্য)

```typescript
// src/features/auth/store/auth.selectors.ts
import { useAuthStore } from './auth.store';

export const useSession = () => useAuthStore((state) => state.session);
export const useAuthUser = () => useAuthStore((state) => state.session?.user ?? null);
export const useIsAuthenticated = () => useAuthStore((state) => !!state.session);
export const useAuthLoading = () => useAuthStore((state) => state.isLoading);
```

Zustand-এ পুরো store subscribe না করে selector দিয়ে নির্দিষ্ট field subscribe করলে অপ্রয়োজনীয় re-render কমে — এটা performance best practice।

## 4. Auth Listener Hook

```typescript
// src/features/auth/hooks/useAuthListener.ts
import { useEffect } from 'react';
import { supabase } from '../../../lib/supabase/client';
import { useAuthStore } from '../store/auth.store';

export function useAuthListener(): void {
  const setSession = useAuthStore((state) => state.setSession);
  const setLoading = useAuthStore((state) => state.setLoading);

  useEffect(() => {
    let isMounted = true;

    supabase.auth.getSession().then(({ data }) => {
      if (!isMounted) return;
      setSession(data.session);
      setLoading(false);
    });

    const { data: subscription } = supabase.auth.onAuthStateChange((_event, session) => {
      setSession(session);
      setLoading(false);
    });

    return () => {
      isMounted = false;
      subscription.subscription.unsubscribe();
    };
  }, [setSession, setLoading]);
}
```

`onAuthStateChange` নিজে থেকেই `SIGNED_IN`, `SIGNED_OUT`, `TOKEN_REFRESHED` সব event-এ fire করে — তাই token refresh হলেও store আপডেট automatic, আলাদা করে কিছু করতে হবে না।

## 5. Root layout-এ wire করো

```typescript
// app/_layout.tsx
import { useAuthListener } from '../src/features/auth/hooks/useAuthListener';
import { useAuthLoading } from '../src/features/auth/store/auth.selectors';
import { Stack } from 'expo-router';
import { ActivityIndicator, View } from 'react-native';

export default function RootLayout() {
  useAuthListener();
  const isLoading = useAuthLoading();

  if (isLoading) {
    return (
      <View className="flex-1 items-center justify-center">
        <ActivityIndicator size="large" />
      </View>
    );
  }

  return <Stack screenOptions={{ headerShown: false }} />;
}
```

## 6. যাচাই করো

আগের ধাপের temporary test code (`console.log`) সরিয়ে ফেলো — এখন এর বদলে যেকোনো স্ক্রিনে:

```typescript
import { useIsAuthenticated, useAuthUser } from '../src/features/auth/store/auth.selectors';

const isAuthenticated = useIsAuthenticated();
const user = useAuthUser();
console.log({ isAuthenticated, email: user?.email });
```

App চালু করলে প্রথমে সংক্ষিপ্ত loading spinner, তারপর `isAuthenticated: false` আসা উচিত (এখনো login UI নাই)।

## এই অংশে যা হলো

- App-wide session state — যেকোনো component থেকে `useIsAuthenticated()` / `useAuthUser()` দিয়ে পড়া যাবে
- Splash → session-check → app এই flow ঠিক হলো, ভুল screen flash হবে না
- Token refresh automatic ভাবে store-এ reflect হবে
