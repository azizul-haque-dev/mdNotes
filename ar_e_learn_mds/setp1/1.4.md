# Step 1.4 — Auth, Error Handling & 404 Middleware

**লক্ষ্য:** Supabase-issued JWT verify করার middleware, global error handler, আর 404 handler।

## ফাইল তৈরি করুন

**`backend/src/middleware/auth.ts`**

```typescript
import type { NextFunction, Request, Response } from "express";
import jwt from "jsonwebtoken";
import jwksClient from "jwks-rsa";

import { env } from "../config/env.js";
import { ApiError } from "../utils/ApiError.js";

export interface AuthenticatedUser {
  id: string; // Supabase auth.users.id (uuid) — the `sub` claim
  email?: string;
  role?: string;
}

declare module "express-serve-static-core" {
  interface Request {
    user?: AuthenticatedUser;
  }
}

// Fetches and caches Supabase's public signing keys. Only re-hits the JWKS
// endpoint when a `kid` shows up that isn't in the cache yet (e.g. after key
// rotation) — see mobile/backend README for the asymmetric-vs-legacy note.
const client = jwksClient({
  jwksUri: env.SUPABASE_JWKS_URI,
  cache: true,
  cacheMaxAge: 10 * 60 * 1000, // matches Supabase's own edge cache TTL
  rateLimit: true,
});

function getSigningKey(
  header: jwt.JwtHeader,
  callback: jwt.SigningKeyCallback,
): void {
  if (!header.kid) {
    callback(new Error('JWT is missing a "kid" header'));
    return;
  }

  client.getSigningKey(header.kid, (err, key) => {
    if (err || !key) {
      callback(err ?? new Error("Unable to resolve signing key"));
      return;
    }
    callback(null, key.getPublicKey());
  });
}

function verifyToken(token: string): Promise<jwt.JwtPayload> {
  return new Promise((resolve, reject) => {
    jwt.verify(
      token,
      getSigningKey,
      { audience: env.SUPABASE_JWT_AUDIENCE, algorithms: ["RS256", "ES256"] },
      (err, decoded) => {
        if (err || !decoded || typeof decoded === "string") {
          reject(err ?? new Error("Invalid token payload"));
          return;
        }
        resolve(decoded);
      },
    );
  });
}

/** Requires a valid Supabase access token; rejects the request otherwise. */
export async function requireAuth(
  req: Request,
  _res: Response,
  next: NextFunction,
) {
  try {
    const header = req.headers.authorization;
    if (!header?.startsWith("Bearer ")) {
      throw ApiError.unauthorized("Missing bearer token");
    }

    const token = header.slice("Bearer ".length);
    const payload = await verifyToken(token);

    if (!payload.sub) {
      throw ApiError.unauthorized("Token missing subject claim");
    }

    req.user = {
      id: payload.sub,
      email: typeof payload.email === "string" ? payload.email : undefined,
      role: typeof payload.role === "string" ? payload.role : undefined,
    };
    next();
  } catch {
    next(ApiError.unauthorized("Invalid or expired token"));
  }
}
```

**`backend/src/middleware/errorHandler.ts`**

```typescript
import type { NextFunction, Request, Response } from "express";
import { ZodError } from "zod";

import { env } from "../config/env.js";
import { logger } from "../config/logger.js";
import { ApiError } from "../utils/ApiError.js";

interface ErrorResponseBody {
  success: false;
  message: string;
  details?: unknown;
  stack?: string;
}

// Express 5 recognizes this by its 4-argument signature — do not remove any param.
export function errorHandler(
  err: unknown,
  req: Request,
  res: Response,
  _next: NextFunction,
): void {
  let statusCode = 500;
  let message = "Internal server error";
  let details: unknown;

  if (err instanceof ZodError) {
    statusCode = 400;
    message = "Validation failed";
    details = err.flatten();
  } else if (err instanceof ApiError) {
    statusCode = err.statusCode;
    message = err.message;
    details = err.details;
  } else if (err instanceof Error) {
    message = env.NODE_ENV === "production" ? message : err.message;
  }

  const isUnexpected = statusCode >= 500;
  logger[isUnexpected ? "error" : "warn"](
    { err, path: req.originalUrl, method: req.method },
    message,
  );

  const body: ErrorResponseBody = { success: false, message };
  if (details !== undefined) body.details = details;
  if (env.NODE_ENV !== "production" && err instanceof Error)
    body.stack = err.stack;

  res.status(statusCode).json(body);
}
```

**`backend/src/middleware/notFound.ts`**

```typescript
import type { NextFunction, Request, Response } from "express";

import { ApiError } from "../utils/ApiError.js";

export function notFoundHandler(
  req: Request,
  _res: Response,
  next: NextFunction,
): void {
  next(ApiError.notFound(`Route not found: ${req.method} ${req.originalUrl}`));
}
```

## আর্কিটেকচার নোট (গুরুত্বপূর্ণ সিদ্ধান্ত)

- **JWKS দিয়ে verify, legacy HS256 secret দিয়ে না।** Supabase-এর বর্তমান recommended পদ্ধতি হলো `/auth/v1/.well-known/jwks.json` থেকে public key fetch+cache করে লোকালি verify করা — প্রতি রিকোয়েস্টে Supabase-এ রাউন্ড ট্রিপ লাগে না। যদি আপনার Supabase প্রজেক্ট এখনো legacy shared secret ব্যবহার করে, হয় migrate করুন (Dashboard → Project Settings → JWT), অথবা এই middleware-এ fallback যোগ করতে হবে।
- **Express 5 async error handling** — `async` route handler-এর ভেতর থ্রো করা error স্বয়ংক্রিয়ভাবে catch হয়ে `errorHandler`-এ যায়, তাই `express-async-errors` প্যাকেজ লাগেনি।
- `errorHandler` অবশ্যই ৪-প্যারামিটার signature-এ থাকতে হবে (`err, req, res, next`) — এটাই Express-এর error-handling middleware চেনার উপায়।
