```typescript
// src/utils/ApiError.ts
export class ApiError extends Error {
  statusCode: number;
  code: string;

  constructor(statusCode: number, code: string, message: string) {
    super(message);
    this.statusCode = statusCode;
    this.code = code;
    Object.setPrototypeOf(this, ApiError.prototype);
  }
}
```

```typescript
// src/utils/asyncHandler.ts
import { Request, Response, NextFunction, RequestHandler } from 'express';

export const asyncHandler = <T extends Request = Request>(
  fn: (req: T, res: Response, next: NextFunction) => Promise<void>,
): RequestHandler => {
  return (req, res, next) => {
    Promise.resolve(fn(req as T, res, next)).catch(next);
  };
};
```

```typescript
// src/middlewares/errorHandler.middleware.ts
import { Request, Response, NextFunction } from 'express';
import { ApiError } from '../utils/ApiError';

export function errorHandler(
  err: unknown,
  _req: Request,
  res: Response,
  _next: NextFunction,
): void {
  if (err instanceof ApiError) {
    res.status(err.statusCode).json({
      success: false,
      error: { code: err.code, message: err.message },
    });
    return;
  }

  console.error('[Unhandled Error]', err);
  res.status(500).json({
    success: false,
    error: { code: 'INTERNAL_ERROR', message: 'Something went wrong' },
  });
}
```

## 3. User Model (Mongoose)

```typescript
// src/features/user/user.model.ts
import { Schema, model, Document } from 'mongoose';

export type NativeLanguage = 'bn' | 'en';

export interface IUser extends Document {
  supabaseUserId: string;
  email: string;
  fullName?: string;
  avatarUrl?: string;
  nativeLanguage: NativeLanguage;
  lastLoginAt: Date;
  createdAt: Date;
  updatedAt: Date;
}

const userSchema = new Schema<IUser>(
  {
    supabaseUserId: { type: String, required: true, unique: true, index: true },
    email: { type: String, required: true, lowercase: true, trim: true },
    fullName: { type: String, trim: true, maxlength: 100 },
    avatarUrl: { type: String, trim: true },
    nativeLanguage: { type: String, enum: ['bn', 'en'], default: 'bn' },
    lastLoginAt: { type: Date, default: Date.now },
  },
  { timestamps: true },
);

export const UserModel = model<IUser>('User', userSchema);
```

`supabaseUserId` কেই primary reference key হিসেবে ব্যবহার করছি — Supabase-এর `auth.users` টেবিলে backend সরাসরি হাত দেবে না, শুধু `sub` claim দিয়ে link রাখছে।

## 5. Repository (শুধু এখানেই Mongoose query থাকবে)

```typescript
// src/features/user/user.repository.ts
import { UserModel, IUser } from './user.model';

export const userRepository = {
  findBySupabaseId(supabaseUserId: string) {
    return UserModel.findOne({ supabaseUserId }).lean();
  },

  create(data: Partial<IUser>) {
    return UserModel.create(data);
  },

  updateBySupabaseId(supabaseUserId: string, data: Partial<IUser>) {
    return UserModel.findOneAndUpdate({ supabaseUserId }, data, { new: true }).lean();
  },
};
```

## 6. Service (business logic)

```typescript
// src/features/user/user.service.ts
import { userRepository } from './user.repository';
import { SyncProfileInput } from './user.validation';
import { SupabaseUserClaims } from '../auth/auth.types';
import { ApiError } from '../../utils/ApiError';

export const userService = {
  async syncProfile(claims: SupabaseUserClaims, input: SyncProfileInput) {
    const existing = await userRepository.findBySupabaseId(claims.sub);

    if (existing) {
      return userRepository.updateBySupabaseId(claims.sub, {
        ...input,
        lastLoginAt: new Date(),
      });
    }

    if (!claims.email) {
      throw new ApiError(400, 'EMAIL_REQUIRED', 'Email is required to create a profile');
    }

    return userRepository.create({
      supabaseUserId: claims.sub,
      email: claims.email,
      fullName: input.fullName,
      avatarUrl: input.avatarUrl,
      nativeLanguage: input.nativeLanguage ?? 'bn',
      lastLoginAt: new Date(),
    });
  },

  async getProfile(supabaseUserId: string) {
    const user = await userRepository.findBySupabaseId(supabaseUserId);

    if (!user) {
      throw new ApiError(404, 'PROFILE_NOT_FOUND', 'Profile not found. Call sync-profile first.');
    }

    return user;
  },
};
```

**কেন first-login-এ auto-create করছি, আলাদা signup endpoint না বানিয়ে:** Supabase-ই signup handle করে ফেলেছে। Mobile app login-এর পর প্রথমবার `sync-profile` call করবে — MongoDB-তে profile না থাকলে তৈরি হবে, থাকলে update হবে (idempotent)। এতে race condition বা duplicate signup logic-এর ঝামেলা নেই।

## 7. Controller

```typescript
// src/features/user/user.controller.ts
import { Response } from 'express';
import { asyncHandler } from '../../utils/asyncHandler';
import { ApiError } from '../../utils/ApiError';
import { userService } from './user.service';
import { syncProfileSchema } from './user.validation';
import { AuthenticatedRequest } from '../auth/auth.types';

export const syncProfile = asyncHandler<AuthenticatedRequest>(async (req, res: Response) => {
  if (!req.user) {
    throw new ApiError(401, 'UNAUTHENTICATED', 'User not authenticated');
  }

  const parsed = syncProfileSchema.safeParse(req.body);
  if (!parsed.success) {
    throw new ApiError(
      400,
      'VALIDATION_ERROR',
      parsed.error.errors.map((e) => e.message).join(', '),
    );
  }

  const profile = await userService.syncProfile(req.user, parsed.data);
  res.status(200).json({ success: true, data: profile });
});

export const getMyProfile = asyncHandler<AuthenticatedRequest>(async (req, res: Response) => {
  if (!req.user) {
    throw new ApiError(401, 'UNAUTHENTICATED', 'User not authenticated');
  }

  const profile = await userService.getProfile(req.user.sub);
  res.status(200).json({ success: true, data: profile });
});
```

```bash
  -H "Authorization: Bearer PASTE_ACCESS_TOKEN_HERE" \
  -H "Content-Type: application/json" \
  -d '{"fullName":"Rahim Uddin","nativeLanguage":"bn"}'

# profile fetch
curl http://localhost:5000/api/v1/profile/me \
  -H "Authorization: Bearer PASTE_ACCESS_TOKEN_HERE"

# validation error টেস্ট (invalid avatarUrl)
curl -X POST http://localhost:5000/api/v1/auth/sync-profile \
  -H "Authorization: Bearer PASTE_ACCESS_TOKEN_HERE" \
  -H "Content-Type: application/json" \
  -d '{"avatarUrl":"not-a-url"}'
```

Expected: প্রথম দুইটায় `success: true` সহ profile data, তৃতীয়টায় `400 VALIDATION_ERROR`।

## এই অংশে যা হলো

- MongoDB `User` model + repository/service/controller/route — clean layer separation
- Idempotent profile sync (auto-create on first login, update on later logins)
- Reusable `ApiError` + `asyncHandler` + global `errorHandler` — এখন থেকে বাকি সব feature-এ এই একই pattern reuse হবে
