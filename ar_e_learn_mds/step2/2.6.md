```typescript
export const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      retry: 2,
```

```typescript
// app/_layout.tsx

  );
}
```

## 5. `authService`-এ OTP methods যোগ করো

```typescript
// src/features/auth/services/auth.service.ts (existing ফাইলে যোগ করো)

function mapAuthError(message: string): string {
  if (message.includes('Invalid login credentials')) return 'ইমেইল অথবা পাসওয়ার্ড ভুল';
  if (message.includes('already registered')) return 'এই ইমেইল দিয়ে আগেই অ্যাকাউন্ট আছে';
  if (message.includes('Password should be at least')) return 'পাসওয়ার্ড কমপক্ষে ৬ ক্যারেক্টার হতে হবে';
  if (message.includes('Token has expired') || message.includes('invalid')) return 'কোড ভুল অথবা মেয়াদ শেষ হয়ে গেছে';
  if (message.includes('rate limit')) return 'অনেকবার চেষ্টা হয়েছে, কিছুক্ষণ পর আবার চেষ্টা করো';
  return 'কিছু একটা সমস্যা হয়েছে, আবার চেষ্টা করো';
}

// ... signInWithPassword, signUpWithPassword আগের মতোই থাকবে, নিচে নতুন যোগ করো:

async sendOtp(email: string): Promise<void> {
  const { error } = await supabase.auth.signInWithOtp({
    email,
    options: { shouldCreateUser: true },
  });
  if (error) throw new Error(mapAuthError(error.message));
},

async verifyOtp(email: string, token: string): Promise<void> {
  const { error } = await supabase.auth.verifyOtp({ email, token, type: 'email' });
  if (error) throw new Error(mapAuthError(error.message));
},
```

(এই দুইটা method existing `authService` object-এর ভেতরে বসাও, `signInWithPassword`-এর পাশে।)

```typescript
export function useSendOtp() {
```

```typescript

import { authService } from '../services/auth.service';

export function useVerifyOtp() {
  return useMutation({
    mutationFn: ({ email, token }: { email: string; token: string }) =>

  });
}
```

## 7. OTP Request Screen

```typescript
// app/(auth)/otp-request.tsx
import { Text, KeyboardAvoidingView, Platform, Pressable } from 'react-native';
import { useForm, Controller } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { Link, router } from 'expo-router';
import { otpRequestSchema, OtpRequestInput } from '../../src/features/auth/schemas/otp.schema';
import { useSendOtp } from '../../src/features/auth/hooks/useSendOtp';
import { TextField } from '../../src/components/ui/TextField';
import { Button } from '../../src/components/ui/Button';

export default function OtpRequestScreen() {
  const sendOtp = useSendOtp();

  const {
    control,
    handleSubmit,
    formState: { errors },
  } = useForm<OtpRequestInput>({
    resolver: zodResolver(otpRequestSchema),
    defaultValues: { email: '' },
  });

  const onSubmit = (values: OtpRequestInput) => {
    sendOtp.mutate(values.email, {
      onSuccess: () => {
        router.push({ pathname: '/(auth)/otp-verify', params: { email: values.email } });
      },
    });
  };

  return (
    <KeyboardAvoidingView
      behavior={Platform.OS === 'ios' ? 'padding' : undefined}
      className="flex-1 justify-center bg-white px-6 dark:bg-gray-950"
    >
      <Text className="mb-2 text-2xl font-bold text-gray-900 dark:text-white">কোড দিয়ে লগইন</Text>
      <Text className="mb-8 text-sm text-gray-500 dark:text-gray-400">
        ইমেইলে ৬ ডিজিটের কোড পাঠানো হবে
      </Text>

      <Controller
        control={control}
        name="email"
        render={({ field: { onChange, onBlur, value } }) => (
          <TextField
            label="ইমেইল"
            value={value}
            onChangeText={onChange}
            onBlur={onBlur}
            error={errors.email?.message}
            keyboardType="email-address"
            autoCapitalize="none"
          />
        )}
      />

      {sendOtp.isError ? <Text className="mb-4 text-sm text-red-500">{sendOtp.error.message}</Text> : null}

      <Button label="কোড পাঠাও" onPress={handleSubmit(onSubmit)} isLoading={sendOtp.isPending} />

      <Link href="/(auth)/login" asChild>
        <Pressable className="mt-6">
          <Text className="text-center text-sm text-gray-600 dark:text-gray-400">
            পাসওয়ার্ড দিয়ে লগইন করবে? <Text className="font-semibold text-emerald-600">এখানে যাও</Text>
          </Text>
        </Pressable>
      </Link>
    </KeyboardAvoidingView>
  );
}
```

## 8. OTP Verify Screen

```typescript
// app/(auth)/otp-verify.tsx
import { Text, KeyboardAvoidingView, Platform } from 'react-native';
import { useForm, Controller } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { useLocalSearchParams } from 'expo-router';
import { otpVerifySchema, OtpVerifyInput } from '../../src/features/auth/schemas/otp.schema';
import { useVerifyOtp } from '../../src/features/auth/hooks/useVerifyOtp';
import { TextField } from '../../src/components/ui/TextField';
import { Button } from '../../src/components/ui/Button';

export default function OtpVerifyScreen() {
  const { email } = useLocalSearchParams<{ email: string }>();
  const verifyOtp = useVerifyOtp();

  const {
    control,
    handleSubmit,
    formState: { errors },
  } = useForm<OtpVerifyInput>({
    resolver: zodResolver(otpVerifySchema),
    defaultValues: { token: '' },
  });

  const onSubmit = (values: OtpVerifyInput) => {
    verifyOtp.mutate({ email, token: values.token });
    // সফল হলে auth listener (2.4) নিজে থেকেই session সেট করবে, redirect (2.8) হবে
  };

  return (
    <KeyboardAvoidingView
      behavior={Platform.OS === 'ios' ? 'padding' : undefined}
      className="flex-1 justify-center bg-white px-6 dark:bg-gray-950"
    >
      <Text className="mb-2 text-2xl font-bold text-gray-900 dark:text-white">কোড দাও</Text>
      <Text className="mb-8 text-sm text-gray-500 dark:text-gray-400">
        {email} এ পাঠানো ৬ ডিজিট কোড দাও
      </Text>

      <Controller
        control={control}
        name="token"
        render={({ field: { onChange, onBlur, value } }) => (
          <TextField
            label="OTP কোড"
            value={value}
            onChangeText={onChange}
            onBlur={onBlur}
            error={errors.token?.message}
            keyboardType="number-pad"
            maxLength={6}
          />
        )}
      />

      {verifyOtp.isError ? (
        <Text className="mb-4 text-sm text-red-500">{verifyOtp.error.message}</Text>
      ) : null}

      <Button label="ভেরিফাই করো" onPress={handleSubmit(onSubmit)} isLoading={verifyOtp.isPending} />
    </KeyboardAvoidingView>
  );
}
```

## 9. যাচাই করো

1. Login screen থেকে "কোড দিয়ে লগইন" (এই লিংক 2.5-এ নাই — চাইলে login screen-এ একটা `Link href="/(auth)/otp-request"` যোগ করো, অথবা সরাসরি URL bar/deep link দিয়ে টেস্ট করো)
2. Email দিয়ে "কোড পাঠাও" চাপো → Supabase inbox/email-এ ৬ ডিজিট কোড আসবে
3. Verify screen-এ কোড বসিয়ে "ভেরিফাই করো" → সফল হলে auth listener session set করবে
4. ভুল কোড দিলে "কোড ভুল অথবা মেয়াদ শেষ" error দেখা উচিত
5. `sendOtp.isPending` / `verifyOtp.isPending` — button-এ spinner দেখা উচিত network call চলাকালীন

---

## এই অংশে যা হলো

- TanStack Query `useMutation` দিয়ে data-fetching — manual `useState`/`useEffect` ছাড়াই `isPending`/`isError`/`error` state পাওয়া যাচ্ছে
- `QueryClientProvider` root-এ wire হলো — এখন থেকে বাকি সব feature (lessons, progress ইত্যাদি)-এও এই একই client reuse হবে
- Passwordless OTP login flow — Supabase নিজেই email deliver করছে, backend touch করা লাগেনি
