## 4. Session-URL parser (Supabase officially recommended pattern)

```typescript
// src/lib/supabase/createSessionFromUrl.ts
import * as QueryParams from 'expo-auth-session/build/QueryParams';
import { Session } from '@supabase/supabase-js';
import { supabase } from './client';

export async function createSessionFromUrl(url: string): Promise<Session | null> {
  const { params, errorCode } = QueryParams.getQueryParams(url);

  if (errorCode) {
    throw new Error(errorCode);
  }

  const { access_token, refresh_token } = params;

  if (!access_token || !refresh_token) {
    return null;
  }

  const { data, error } = await supabase.auth.setSession({ access_token, refresh_token });
  if (error) throw error;

  return data.session;
}
```

## 5. `authService`-এ Google method যোগ করো

```typescript
// src/features/auth/services/auth.service.ts (existing ফাইলে যোগ করো)
import * as WebBrowser from 'expo-web-browser';
import * as Linking from 'expo-linking';
import { createSessionFromUrl } from '../../../lib/supabase/createSessionFromUrl';

// ... signInWithPassword, signUpWithPassword, sendOtp, verifyOtp আগের মতোই, নিচে যোগ করো:

async signInWithGoogle(): Promise<void> {
  const redirectTo = Linking.createURL('auth-callback');

  const { data, error } = await supabase.auth.signInWithOAuth({
    provider: 'google',
    options: { redirectTo, skipBrowserRedirect: true },
  });

  if (error || !data?.url) {
    throw new Error(mapAuthError(error?.message ?? 'গুগল লগইন শুরু করা যায়নি'));
  }

  const result = await WebBrowser.openAuthSessionAsync(data.url, redirectTo);

  if (result.type !== 'success' || !result.url) {
    throw new Error('গুগল লগইন বাতিল করা হয়েছে');
  }

  const session = await createSessionFromUrl(result.url);
  if (!session) {
    throw new Error('গুগল থেকে session পাওয়া যায়নি, আবার চেষ্টা করো');
  }
},
```

## 7. Mutation hook

```typescript

import { useMutation } from '@tanstack/react-query';
import { authService } from '../services/auth.service';

export function useGoogleSignIn() {
  return useMutation({
    mutationFn: () => authService.signInWithGoogle(),
  });
}
```

## 8. Google Sign-In Button component

```typescript
// src/features/auth/components/GoogleSignInButton.tsx
import { Pressable, Text, ActivityIndicator } from 'react-native';
import { useGoogleSignIn } from '../hooks/useGoogleSignIn';

export function GoogleSignInButton() {
  const googleSignIn = useGoogleSignIn();

  return (
    <Pressable
      onPress={() => googleSignIn.mutate()}
      disabled={googleSignIn.isPending}
      className="flex-row items-center justify-center rounded-xl border border-gray-300 py-3.5 dark:border-gray-700"
    >
      {googleSignIn.isPending ? (
        <ActivityIndicator />
      ) : (
        <Text className="text-base font-semibold text-gray-900 dark:text-white">
          Google দিয়ে লগইন
        </Text>
      )}
    </Pressable>
  );
}
```

## 10. যাচাই করো

⚠️ **Google OAuth deep link `Expo Go`-তে সঠিকভাবে কাজ নাও করতে পারে (custom scheme লাগে)।** Dev client build দিয়ে টেস্ট করো:

```bash
npx expo run:android
# অথবা
npx expo run:ios
```

1. "Google দিয়ে লগইন" চাপো → browser খুলে Google account picker আসা উচিত
2. Google account বেছে নেওয়ার পর app-এ ফিরে আসা উচিত
3. Auth listener (2.4) session set করবে → `isAuthenticated: true` হওয়া উচিত
4. Cancel করলে "গুগল লগইন বাতিল করা হয়েছে" error দেখা উচিত

## এই অংশে যা হলো

- Email + OTP + Google — তিনটা login method-ই এখন কাজ করছে, একই `authService`/auth-store/listener architecture-এর মধ্যে দিয়ে
- OAuth token URL parsing officially recommended utility দিয়ে — নিজে regex লেখার ঝুঁকি এড়ানো হলো
- সব method একই jায়গায় গিয়ে মিশছে (`onAuthStateChange` listener) — তাই কোন method দিয়ে login হলো তার উপর ভিত্তি করে বাকি app-এর কোনো আলাদা লজিক লাগবে না
