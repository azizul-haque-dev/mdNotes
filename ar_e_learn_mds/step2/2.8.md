## 3. Axios client + auto token injection

```typescript
// src/lib/api/client.ts
import axios from 'axios';
import { supabase } from '../supabase/client';

const API_BASE_URL = process.env.EXPO_PUBLIC_API_BASE_URL;

if (!API_BASE_URL) {
  throw new Error('Missing EXPO_PUBLIC_API_BASE_URL in .env');
}

export const apiClient = axios.create({
  baseURL: API_BASE_URL,
  timeout: 15000,
});

apiClient.interceptors.request.use(async (config) => {
  const { data } = await supabase.auth.getSession();
  const accessToken = data.session?.access_token;

  if (accessToken) {
    config.headers.Authorization = `Bearer ${accessToken}`;
  }

  return config;
});

apiClient.interceptors.response.use(
  (response) => response,
  (error) => {
    if (!error.response) {
      return Promise.reject(new Error('ইন্টারনেট কানেকশন চেক করো'));
    }

    const message = error.response.data?.error?.message ?? 'কিছু একটা সমস্যা হয়েছে';
    return Promise.reject(new Error(message));
  },
);
```

প্রতিটা request-এ manual token বসানো লাগবে না — এই interceptor সব জায়গায় reuse হবে।

## 4. Profile service

```typescript
// src/features/user/services/profile.service.ts
import { apiClient } from '../../../lib/api/client';

export interface UserProfile {
  _id: string;
  supabaseUserId: string;
  email: string;
  fullName?: string;
  avatarUrl?: string;
  nativeLanguage: 'bn' | 'en';
  lastLoginAt: string;
}

interface ApiSuccessResponse<T> {
  success: true;
  data: T;
}

export const profileService = {
  async syncProfile(): Promise<UserProfile> {
    const response = await apiClient.post<ApiSuccessResponse<UserProfile>>(
      '/api/v1/auth/sync-profile',
      {},
    );
    return response.data.data;
  },

  async getMyProfile(): Promise<UserProfile> {
    const response = await apiClient.get<ApiSuccessResponse<UserProfile>>('/api/v1/profile/me');
    return response.data.data;
  },
};
```

```typescript
// src/features/user/hooks/useSyncProfile.ts
import { useMutation } from '@tanstack/react-query';
import { profileService } from '../services/profile.service';

export function useSyncProfile() {
  return useMutation({
    mutationFn: () => profileService.syncProfile(),
  });
}
```

## 5. `useAuthListener`-এ sync trigger যোগ করো

```typescript
// src/features/auth/hooks/useAuthListener.ts (existing ফাইল আপডেট করো)
import { useEffect } from 'react';
import { supabase } from '../../../lib/supabase/client';
import { useAuthStore } from '../store/auth.store';
import { useSyncProfile } from '../../user/hooks/useSyncProfile';

export function useAuthListener(): void {
  const setSession = useAuthStore((state) => state.setSession);
  const setLoading = useAuthStore((state) => state.setLoading);
  const syncProfile = useSyncProfile();

  useEffect(() => {
    let isMounted = true;

    supabase.auth.getSession().then(({ data }) => {
      if (!isMounted) return;
      setSession(data.session);
      setLoading(false);
    });

    const { data: subscription } = supabase.auth.onAuthStateChange((event, session) => {
      setSession(session);
      setLoading(false);

      // password/OTP/Google — যেভাবেই login হোক, এই event একবার fire হয়
      if (event === 'SIGNED_IN') {
        syncProfile.mutate();
      }
    });

    return () => {
      isMounted = false;
      subscription.subscription.unsubscribe();
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [setSession, setLoading]);
}
```

`syncProfile` কে dependency array-তে ইচ্ছাকৃতভাবে রাখিনি — `useMutation` প্রতি render-এ নতুন object দেয়, দিলে listener বারবার re-subscribe হতো।

## 9. Root `index.tsx` replace করো (temporary redirect বাদ)

```typescript
// app/index.tsx
import { Redirect } from 'expo-router';
import { useIsAuthenticated } from '../src/features/auth/store/auth.selectors';

export default function Index() {
  const isAuthenticated = useIsAuthenticated();
  return <Redirect href={isAuthenticated ? '/(app)' : '/(auth)/login'} />;
}
```

Root layout (2.4-এ বানানো) ইতিমধ্যে `isLoading` শেষ না হওয়া পর্যন্ত `Stack` render করে না — তাই এখানে flicker হওয়ার সুযোগ নেই, `Index` render হওয়ার সময় auth state ঠিকভাবে settle হয়েই থাকে।

## 10. যাচাই করো (পুরো auth flow end-to-end)

1. Backend চালাও (`npm run dev`) — নিশ্চিত করো তোমার device/emulator থেকে সেই IP-তে reach করা যাচ্ছে
2. App চালাও → login screen দেখা উচিত
3. Email/Password অথবা OTP অথবা Google — যেকোনো একটা দিয়ে login করো
4. Backend terminal-এ `POST /api/v1/auth/sync-profile` লগ দেখা উচিত
5. App home screen-এ redirect হওয়া উচিত, তোমার email দেখা উচিত
6. MongoDB compass/shell-এ চেক করো — নতুন `User` document তৈরি হয়েছে কিনা
7. "লগ আউট" চাপো → login screen-এ ফিরে আসা উচিত
8. আবার login করো → এবার `sync-profile` existing profile **update** করবে (নতুন তৈরি করবে না), `lastLoginAt` বদলে যাওয়া উচিত
