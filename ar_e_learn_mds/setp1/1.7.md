## ফাইল তৈরি করুন

**`mobile/src/core/config/env.ts`**

```typescript
import { z } from "zod";

/**
 * Expo only exposes variables prefixed with EXPO_PUBLIC_ to client code
 * (see .env.example). Never put secrets here — anything referenced through
 * `process.env.EXPO_PUBLIC_*` is bundled into the client binary.
 */
const envSchema = z.object({
  EXPO_PUBLIC_API_BASE_URL: z.string().url(),
  EXPO_PUBLIC_SUPABASE_URL: z.string().url(),
  EXPO_PUBLIC_SUPABASE_ANON_KEY: z.string().min(1),
});

function loadEnv() {
  const parsed = envSchema.safeParse({
    EXPO_PUBLIC_API_BASE_URL: process.env.EXPO_PUBLIC_API_BASE_URL,
    EXPO_PUBLIC_SUPABASE_URL: process.env.EXPO_PUBLIC_SUPABASE_URL,
    EXPO_PUBLIC_SUPABASE_ANON_KEY: process.env.EXPO_PUBLIC_SUPABASE_ANON_KEY,
  });

  if (!parsed.success) {
    // Fail fast and loud in dev rather than shipping a build with a broken API URL.
    throw new Error(
      `Invalid environment configuration: ${parsed.error.issues
        .map((issue) => issue.path.join("."))
        .join(", ")}. Check mobile/.env against mobile/.env.example.`,
    );
  }

  return parsed.data;
}

export const env = loadEnv();
```

**`mobile/src/core/config/supabase.ts`**

```typescript
import AsyncStorage from "@react-native-async-storage/async-storage";
import { createClient } from "@supabase/supabase-js";
import * as aesjs from "aes-js";
import * as SecureStore from "expo-secure-store";
import "react-native-get-random-values";

import { env } from "../config/env";

/**
 * Supabase sessions (access + refresh token, user metadata) routinely exceed
 * SecureStore's ~2048-byte per-value limit. The pattern Supabase recommends
 * for Expo: keep a small AES-256 key in SecureStore (hardware-backed
 * Keychain/Keystore), and use it to encrypt the actual session blob before
 * writing it to AsyncStorage. AsyncStorage alone would be plaintext-on-disk,
 * which is why we don't use it directly for anything auth-related.
 */
class LargeSecureStore {
  private async getEncryptionKey(keyName: string): Promise<Uint8Array> {
    const existing = await SecureStore.getItemAsync(keyName);
    if (existing) {
      return aesjs.utils.hex.toBytes(existing);
    }

    const newKey = crypto.getRandomValues(new Uint8Array(32));
    await SecureStore.setItemAsync(keyName, aesjs.utils.hex.fromBytes(newKey));
    return newKey;
  }

  async getItem(key: string): Promise<string | null> {
    const encrypted = await AsyncStorage.getItem(key);
    if (!encrypted) return null;

    const keyBytes = await this.getEncryptionKey(`${key}-secure-key`);
    const counter = new aesjs.Counter(1);
    const cipher = new aesjs.ModeOfOperation.ctr(keyBytes, counter);
    const decryptedBytes = cipher.decrypt(aesjs.utils.hex.toBytes(encrypted));
    return aesjs.utils.utf8.fromBytes(decryptedBytes);
  }

  async setItem(key: string, value: string): Promise<void> {
    const keyBytes = await this.getEncryptionKey(`${key}-secure-key`);
    const counter = new aesjs.Counter(1);
    const cipher = new aesjs.ModeOfOperation.ctr(keyBytes, counter);
    const encryptedBytes = cipher.encrypt(aesjs.utils.utf8.toBytes(value));
    await AsyncStorage.setItem(key, aesjs.utils.hex.fromBytes(encryptedBytes));
  }

  async removeItem(key: string): Promise<void> {
    await AsyncStorage.removeItem(key);
    await SecureStore.deleteItemAsync(`${key}-secure-key`);
  }
}

export const supabase = createClient(
  env.EXPO_PUBLIC_SUPABASE_URL,
  env.EXPO_PUBLIC_SUPABASE_ANON_KEY,
  {
    auth: {
      storage: new LargeSecureStore(),
      autoRefreshToken: true,
      persistSession: true,
      detectSessionInUrl: false,
    },
  },
);
```

**`mobile/src/core/network/queryClient.ts`**

```typescript
import { QueryClient } from "@tanstack/react-query";

/**
 * A single shared QueryClient instance.
 *
 * Defaults are tuned for an offline-first mobile app:
 * - retry is capped (mobile networks fail often; we don't want to hammer them)
 * - staleTime is non-zero so screens don't refetch every focus while offline
 * - gcTime is generous so cached data survives brief app backgrounding
 *
 * Persistence (so cached data survives app restarts while offline) will be
 * wired up together with the SQLite layer in the "Offline sync" milestone.
 */
export const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      retry: 2,
      staleTime: 1000 * 60 * 5, // 5 minutes
      gcTime: 1000 * 60 * 60 * 24, // 24 hours
      refetchOnReconnect: true,
      refetchOnWindowFocus: false,
    },
    mutations: {
      retry: 1,
    },
  },
});
```

**`mobile/src/core/network/api.ts`**

```typescript
import axios, { AxiosError } from "axios";

import { env } from "../config/env";
import { supabase } from "../config/supabase";

export const api = axios.create({
  baseURL: env.EXPO_PUBLIC_API_BASE_URL,
  timeout: 15_000,
});

api.interceptors.request.use(async (config) => {
  const {
    data: { session },
  } = await supabase.auth.getSession();

  if (session?.access_token) {
    config.headers.Authorization = `Bearer ${session.access_token}`;
  }

  return config;
});

/**
 * Normalizes backend error responses (see backend's errorHandler.ts, which
 * always returns `{ success: false, message, details? }`) into a single
 * shape the UI layer can rely on regardless of what failed.
 */
export interface ApiErrorShape {
  message: string;
  status?: number;
  details?: unknown;
}

api.interceptors.response.use(
  (response) => response,
  (error: AxiosError<{ message?: string; details?: unknown }>) => {
    const normalized: ApiErrorShape = {
      message:
        error.response?.data?.message ??
        error.message ??
        "Something went wrong",
      status: error.response?.status,
      details: error.response?.data?.details,
    };
    return Promise.reject(normalized);
  },
);
```

## আর্কিটেকচার নোট (গুরুত্বপূর্ণ সিদ্ধান্ত)

- **`LargeSecureStore` প্যাটার্ন** — Supabase session (access + refresh token + user metadata) সাধারণত SecureStore-এর ~2048-byte per-value লিমিট ছাড়িয়ে যায়। তাই ছোট একটা AES-256 key SecureStore-এ (hardware-backed Keychain/Keystore) রাখা হয়, আর সেটা দিয়ে আসল session blob এনক্রিপ্ট করে AsyncStorage-এ লেখা হয়। এটা Supabase-এরই অফিসিয়াল ডকুমেন্টেড পদ্ধতি Expo অ্যাপের জন্য — শর্টকাট না।
- `api.ts`-এর axios interceptor প্রতি রিকোয়েস্টে current Supabase session token attach করে। Offline queueing/retry এখনো নেই — সেটা "Offline sync" milestone-এ (Step 4) আসবে।
