```typescript
import { z } from 'zod';

export const signupSchema = z
  .object({
    email: z.string().trim().toLowerCase().email('সঠিক ইমেইল দাও'),

  });

export type SignupInput = z.infer<typeof signupSchema>;
```

## 3. Auth Service (Supabase call এখান থেকেই হবে, component থেকে না)

```typescript
// src/features/auth/services/auth.service.ts
import { supabase } from '../../../lib/supabase/client';

interface AuthResult {
  success: boolean;
  errorMessage?: string;
  needsEmailConfirmation?: boolean;
}

function mapAuthError(message: string): string {
  if (message.includes('Invalid login credentials')) return 'ইমেইল অথবা পাসওয়ার্ড ভুল';
  if (message.includes('already registered')) return 'এই ইমেইল দিয়ে আগেই অ্যাকাউন্ট আছে';
  if (message.includes('Password should be at least'))
    return 'পাসওয়ার্ড কমপক্ষে ৬ ক্যারেক্টার হতে হবে';
  if (message.includes('rate limit')) return 'অনেকবার চেষ্টা হয়েছে, কিছুক্ষণ পর আবার চেষ্টা করো';
  return 'কিছু একটা সমস্যা হয়েছে, আবার চেষ্টা করো';
}

export const authService = {
  async signInWithPassword(email: string, password: string): Promise<AuthResult> {
    const { error } = await supabase.auth.signInWithPassword({ email, password });
    if (error) return { success: false, errorMessage: mapAuthError(error.message) };
    return { success: true };
  },

  async signUpWithPassword(email: string, password: string): Promise<AuthResult> {
    const { data, error } = await supabase.auth.signUp({ email, password });
    if (error) return { success: false, errorMessage: mapAuthError(error.message) };

    // Supabase project-এ email confirmation "on" থাকলে signUp session দেয় না
    return { success: true, needsEmailConfirmation: !data.session };
  },
};
```

```typescript
import { Pressable, Text, ActivityIndicator } from 'react-native';

interface ButtonProps {
  label: string;
  onPress: () => void;
  isLoading?: boolean;
  disabled?: boolean;
}

export function Button({ label, onPress, isLoading, disabled }: ButtonProps) {
  const isDisabled = disabled || isLoading;

  return (
    <Pressable
      onPress={onPress}
      disabled={isDisabled}
      className={`items-center justify-center rounded-xl py-3.5 ${
        isDisabled ? 'bg-emerald-300' : 'bg-emerald-600 active:bg-emerald-700'
      }`}
    >
      {isLoading ? (
        <ActivityIndicator color="#fff" />
      ) : (
        <Text className="text-base font-semibold text-white">{label}</Text>
      )}
    </Pressable>
  );
}
```

## 6. Login Screen

```typescript
// app/(auth)/login.tsx
import { useState } from 'react';
import { View, Text, KeyboardAvoidingView, Platform, Pressable } from 'react-native';
import { useForm, Controller } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { Link } from 'expo-router';
import { loginSchema, LoginInput } from '../../src/features/auth/schemas/login.schema';
import { authService } from '../../src/features/auth/services/auth.service';
import { TextField } from '../../src/components/ui/TextField';
import { Button } from '../../src/components/ui/Button';

export default function LoginScreen() {
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [serverError, setServerError] = useState<string | null>(null);

  const {
    control,
    handleSubmit,
    formState: { errors },
  } = useForm<LoginInput>({
    resolver: zodResolver(loginSchema),
    defaultValues: { email: '', password: '' },
  });

  const onSubmit = async (values: LoginInput) => {
    setServerError(null);
    setIsSubmitting(true);
    const result = await authService.signInWithPassword(values.email, values.password);
    setIsSubmitting(false);

    if (!result.success) {
      setServerError(result.errorMessage ?? 'লগইন ব্যর্থ হয়েছে');
      return;
    }
    // সফল login-এ auth listener (2.4) নিজে থেকেই session আপডেট করবে —
    // এখানে ম্যানুয়াল navigation লাগবে না
  };

  return (
    <KeyboardAvoidingView
      behavior={Platform.OS === 'ios' ? 'padding' : undefined}
      className="flex-1 justify-center bg-white px-6 dark:bg-gray-950"
    >
      <Text className="mb-8 text-2xl font-bold text-gray-900 dark:text-white">লগইন করো</Text>

      <Controller
        control={control}
        name="email"
        render={({ field: { onChange, onBlur, value } }) => (
          <TextField
            label="ইমেইল"
            value={value}
            onChangeText={onChange}
            onBlur={onBlur}
            error={errors.email?.message}
            keyboardType="email-address"
            autoCapitalize="none"
          />
        )}
      />

      <Controller
        control={control}
        name="password"
        render={({ field: { onChange, onBlur, value } }) => (
          <TextField
            label="পাসওয়ার্ড"
            value={value}
            onChangeText={onChange}
            onBlur={onBlur}
            error={errors.password?.message}
            secureTextEntry
          />
        )}
      />

      {serverError ? <Text className="mb-4 text-sm text-red-500">{serverError}</Text> : null}

      <Button label="লগইন" onPress={handleSubmit(onSubmit)} isLoading={isSubmitting} />

      <Link href="/(auth)/signup" asChild>
        <Pressable className="mt-6">
          <Text className="text-center text-sm text-gray-600 dark:text-gray-400">
            অ্যাকাউন্ট নাই? <Text className="font-semibold text-emerald-600">সাইন আপ করো</Text>
          </Text>
        </Pressable>
      </Link>
    </KeyboardAvoidingView>
  );
}
```

## 7. Signup Screen

```typescript
// app/(auth)/signup.tsx
import { useState } from 'react';
import { Text, KeyboardAvoidingView, Platform, Pressable, Alert } from 'react-native';
import { useForm, Controller } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { Link, router } from 'expo-router';
import { signupSchema, SignupInput } from '../../src/features/auth/schemas/signup.schema';
import { authService } from '../../src/features/auth/services/auth.service';
import { TextField } from '../../src/components/ui/TextField';
import { Button } from '../../src/components/ui/Button';

export default function SignupScreen() {
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [serverError, setServerError] = useState<string | null>(null);

  const {
    control,
    handleSubmit,
    formState: { errors },
  } = useForm<SignupInput>({
    resolver: zodResolver(signupSchema),
    defaultValues: { email: '', password: '', confirmPassword: '' },
  });

  const onSubmit = async (values: SignupInput) => {
    setServerError(null);
    setIsSubmitting(true);
    const result = await authService.signUpWithPassword(values.email, values.password);
    setIsSubmitting(false);

    if (!result.success) {
      setServerError(result.errorMessage ?? 'সাইন আপ ব্যর্থ হয়েছে');
      return;
    }

    if (result.needsEmailConfirmation) {
      Alert.alert('ইমেইল চেক করো', 'অ্যাকাউন্ট ভেরিফাই করার লিংক পাঠানো হয়েছে');
      router.replace('/(auth)/login');
      return;
    }
    // session তৈরি হয়ে গেলে auth listener নিজেই redirect করবে
  };

  return (
    <KeyboardAvoidingView
      behavior={Platform.OS === 'ios' ? 'padding' : undefined}
      className="flex-1 justify-center bg-white px-6 dark:bg-gray-950"
    >
      <Text className="mb-8 text-2xl font-bold text-gray-900 dark:text-white">অ্যাকাউন্ট বানাও</Text>

      <Controller
        control={control}
        name="email"
        render={({ field: { onChange, onBlur, value } }) => (
          <TextField
            label="ইমেইল"
            value={value}
            onChangeText={onChange}
            onBlur={onBlur}
            error={errors.email?.message}
            keyboardType="email-address"
            autoCapitalize="none"
          />
        )}
      />

      <Controller
        control={control}
        name="password"
        render={({ field: { onChange, onBlur, value } }) => (
          <TextField
            label="পাসওয়ার্ড"
            value={value}
            onChangeText={onChange}
            onBlur={onBlur}
            error={errors.password?.message}
            secureTextEntry
          />
        )}
      />

      <Controller
        control={control}
        name="confirmPassword"
        render={({ field: { onChange, onBlur, value } }) => (
          <TextField
            label="পাসওয়ার্ড আবার লিখো"
            value={value}
            onChangeText={onChange}
            onBlur={onBlur}
            error={errors.confirmPassword?.message}
            secureTextEntry
          />
        )}
      />

      {serverError ? <Text className="mb-4 text-sm text-red-500">{serverError}</Text> : null}

      <Button label="সাইন আপ" onPress={handleSubmit(onSubmit)} isLoading={isSubmitting} />

      <Link href="/(auth)/login" asChild>
        <Pressable className="mt-6">
          <Text className="text-center text-sm text-gray-600 dark:text-gray-400">
            অ্যাকাউন্ট আছে? <Text className="font-semibold text-emerald-600">লগইন করো</Text>
          </Text>
        </Pressable>
      </Link>
    </KeyboardAvoidingView>
  );
}
```

## 9. যাচাই করো

1. App চালাও → login screen দেখা উচিত
2. ভুল email/password দিয়ে submit করো → Bangla error message দেখা উচিত
3. "সাইন আপ করো" চেপে নতুন email/password দিয়ে account বানাও
4. Supabase project-এ email confirmation off থাকলে সাথে সাথেই logged-in state-এ যাওয়া উচিত (auth listener কাজ করলে `isAuthenticated: true` হবে)
5. On থাকলে "ইমেইল চেক করো" alert আসবে

## এই অংশে যা হলো

- Email/Password দিয়ে পুরো login + signup flow — validation, loading state, Bangla error handling সহ
- Reusable `TextField`/`Button` — পরের সব ফর্মে (OTP, profile, quiz) reuse হবে
- Auth service layer আলাদা — component-এ Supabase call directly নাই
